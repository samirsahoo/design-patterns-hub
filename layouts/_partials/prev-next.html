{{- /* Previous/next within the current section, in weight order. Computed explicitly because Hugo's .Prev/.Next direction depends on default sort. */}} {{- $pages := .CurrentSection.RegularPages.ByWeight }} {{- $prev := false }} {{- $next := false }} {{- range $i, $p := $pages }} {{- if eq $p.RelPermalink $.RelPermalink }} {{- if gt $i 0 }}{{ $prev = index $pages (sub $i 1) }}{{ end }} {{- if lt (add $i 1) (len $pages) }}{{ $next = index $pages (add $i 1) }}{{ end }} {{- end }} {{- end }} {{- if or $prev $next }}
{{- with $prev }} ← Previous {{ .LinkTitle }} {{- else }}{{ end }} {{- with $next }} Next → {{ .LinkTitle }} {{- end }}
{{- end }}
